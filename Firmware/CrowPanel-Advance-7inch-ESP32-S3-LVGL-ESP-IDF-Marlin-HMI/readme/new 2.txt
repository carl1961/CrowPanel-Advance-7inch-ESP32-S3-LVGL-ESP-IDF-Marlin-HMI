void app_main(void)
{
    // 1. Create I2C bus
    i2c_master_bus_config_t bus_cfg = {
        .i2c_port = I2C_NUM_0,
        .scl_io_num = GPIO_NUM_16,
        .sda_io_num = GPIO_NUM_15,
        .clk_source = I2C_CLK_SRC_DEFAULT,
        .glitch_ignore_cnt = 7,
        .flags.enable_internal_pullup = true,
    };
    ESP_ERROR_CHECK(i2c_new_master_bus(&bus_cfg, &i2c_bus));

    // 2. Init STC8 component
    i2c_device_config_t dev_cfg = {
        .dev_addr_length = I2C_ADDR_BIT_LEN_7,
        .device_address  = 0x30,
        .scl_speed_hz    = 400000,
    };
    ESP_ERROR_CHECK(stc8h1k28_init(i2c_bus, &stc8, &dev_cfg));

    // 3. Test buzzer
    stc8h1k28_buzzer_on(stc8);
    vTaskDelay(pdMS_TO_TICKS(500));
    stc8h1k28_buzzer_off(stc8);

    // 4. Init display & LVGL
    init_display();
    ESP_LOGI(TAG, "Start LVGL");

    // 5. Create UI
    if (lvgl_port_lock(-1)) {
        ui_init();
        lvgl_port_unlock();
        ESP_LOGI(TAG, "UI initialized successfully!");
    } else {
        ESP_LOGE(TAG, "Failed to lock LVGL port for UI initialization");
    }
}
