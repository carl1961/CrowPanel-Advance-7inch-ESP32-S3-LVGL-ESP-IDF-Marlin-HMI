#include <stdio.h>

#define MARLIN_RX_PIN       GPIO_NUM_19
#define SERIAL_BUF_SIZE     1024
#define MAX_GCODE_LEN       64
#define COMMAND_QUEUE_LEN   20

#define MAX_FILES           50
#define FILE_NAME_LEN       64

// ─── Types ───────────────────────────────────────────────────────────────────

typedef struct {

} log_item_t;

// ─── Globals ─────────────────────────────────────────────────────────────────

QueueHandle_t marlin_cmd_queue = NULL;

static stc8h1k28_handle_t stc8 = NULL;

bool marlin_connected = false;
bool isPrinting = false;
uint32_t last_marlin_reply_ms = 0;

// M20 file list state

extern "C" void update_clock_labels_timer(lv_timer_t *timer);

// ─── Log queue ───────────────────────────────────────────────────────────────

void log_queue_init(void) {
    log_queue = xQueueCreate(LOG_QUEUE_LEN, sizeof(log_item_t));
}

void append_print_log(const char *line) {
    if (log_queue == NULL) return;
    log_item_t item;
    snprintf(item.line, LOG_LINE_MAX, "%s", line);
    xQueueSend(log_queue, &item, 0);
}

// ─── G-code send ─────────────────────────────────────────────────────────────

extern "C" void send_gcode(const char *command) {
    if (!command) return;
    if (marlin_cmd_queue != NULL) {
        xQueueSend(marlin_cmd_queue, command, 0);
    }
    if (log_queue != NULL) {
        log_item_t item;
        snprintf(item.line, LOG_LINE_MAX, "> %s", command);
        xQueueSend(log_queue, &item, 0);
    }
}

// ─── File browser ────────────────────────────────────────────────────────────

static void update_file_browser_ui(void) {
    if (log_queue == NULL) return;

    log_item_t item;
    snprintf(item.line, LOG_LINE_MAX, "LIST_CLEAR");
    xQueueSend(log_queue, &item, 0);

    for (int i = 0; i < file_count; i++) {
        snprintf(item.line, LOG_LINE_MAX, "LIST_ADD:%.63s", file_list[i]);
        xQueueSend(log_queue, &item, 0);
    }
}

// ← ADDED: click handler for file list items
static void file_browser_click_cb(lv_event_t *e) {
    lv_obj_t *btn = lv_event_get_current_target(e);
    if (!btn) return;
    lv_obj_t *label = lv_obj_get_child(btn, 0);
    if (!label) return;
    const char *filename = lv_label_get_text(label);
    if (!filename || strlen(filename) == 0) return;

    char cmd[80];
    snprintf(cmd, sizeof(cmd), "M23 %s", filename);
    send_gcode(cmd);
    isPrinting = true;
}

// ─── Marlin line parser ──────────────────────────────────────────────────────

void parse_marlin_line(const char *line) {
    if (line == NULL || strlen(line) == 0) return;

    // Temperature updates
    float hcur = -1, bcur = -1, dcur = -1;
    const char *tPos = strstr(line, "T:");
    if (tPos) sscanf(tPos, "T:%f", &hcur);
    const char *bPos = strstr(line, "B:");
    if (bPos) sscanf(bPos, "B:%f", &bcur);
    const char *cPos = strstr(line, "C:");
    if (cPos) sscanf(cPos, "C:%f", &dcur);

    char buf[16];
    if (hcur >= 0 && objects.nozzle_temp_main_screen) {
        snprintf(buf, sizeof(buf), "%.0f°", hcur);
        lv_label_set_text(objects.nozzle_temp_main_screen, buf);
    }
    if (bcur >= 0 && objects.bed_temp_main_screen) {
        snprintf(buf, sizeof(buf), "%.0f°", bcur);
        lv_label_set_text(objects.bed_temp_main_screen, buf);
    }
    if (dcur >= 0 && objects.drivers_temp_main_screen) {
        snprintf(buf, sizeof(buf), "%.0f°", dcur);
        lv_label_set_text(objects.drivers_temp_main_screen, buf);
    }

    // M20 file list parsing
    if (strcmp(line, "Begin file list") == 0) {
        file_count = 0;
        capturing_files = true;
        return;
    }
    if (strcmp(line, "End file list") == 0) {
        capturing_files = false;
        update_file_browser_ui();
        return;
    }
    if (capturing_files && file_count < MAX_FILES) {
        const char *name = line;

        char *colon = strchr(line, ':');
        if (colon && (colon - line) < 3) {
            name = colon + 1;
        }

        char *tmp = (char *)name;
        char *last_space = strrchr(tmp, ' ');
        if (last_space) {
            char *size_str = last_space + 1;
            bool is_size = (size_str[0] != '\0');
            for (char *p = size_str; *p; p++) {
                if (*p < '0' || *p > '9') { is_size = false; break; }
            }
            if (is_size) *last_space = '\0';
        }

        snprintf(file_list[file_count], FILE_NAME_LEN, "%.63s", tmp);
        file_count++;
    }

    // ← ADDED: detect print start / completion from Marlin
    if (strstr(line, "Start SD print job:")) {
        isPrinting = true;
    }
    if (strstr(line, "Done printing file")) {
        isPrinting = false;
    }
}

// ─── Tasks ───────────────────────────────────────────────────────────────────

static void marlin_tx_task(void *arg) {
    char cmd[MAX_GCODE_LEN];
    while (1) {
        if (xQueueReceive(marlin_cmd_queue, cmd, portMAX_DELAY)) {
            size_t len = strlen(cmd);
            if (cmd[len - 1] != '\n') {
                cmd[len] = '\n';
                len++;
            }
            uart_write_bytes(MARLIN_UART_NUM, cmd, len);
            uart_wait_tx_done(MARLIN_UART_NUM, pdMS_TO_TICKS(100));
        }
    }
}

static void marlin_rx_task(void *arg) {
    uint8_t rx_buf[256];
    static char line_accumulator[512];
    static size_t line_pos = 0;

    while (1) {
        int len = uart_read_bytes(MARLIN_UART_NUM, rx_buf, sizeof(rx_buf) - 1, pdMS_TO_TICKS(20));
        if (len > 0) {
            last_marlin_reply_ms = xTaskGetTickCount() * portTICK_PERIOD_MS;
            marlin_connected = true;

            for (int i = 0; i < len; i++) {
                char c = (char)rx_buf[i];
                if (c == '\n') {
                    line_accumulator[line_pos] = '\0';
                    if (line_pos > 0) {
                        parse_marlin_line(line_accumulator);

                        if (strstr(line_accumulator, "T:") == NULL &&
                            strcmp(line_accumulator, "ok") != 0 &&
                            strncmp(line_accumulator, "ok ", 3) != 0 &&
                            !capturing_files &&
                            strcmp(line_accumulator, "Begin file list") != 0 &&
                            strcmp(line_accumulator, "End file list") != 0) {
                            append_print_log(line_accumulator);
                        }
                    }
                    line_pos = 0;
                } else if (c != '\r' && line_pos < sizeof(line_accumulator) - 1) {
                    line_accumulator[line_pos++] = c;
                }
            }
        }
        vTaskDelay(pdMS_TO_TICKS(10));
    }
}

static void status_poller_task(void *arg) {
    while (1) {
        if (isPrinting) send_gcode("M27");
        vTaskDelay(pdMS_TO_TICKS(2000));
    }
}

static void watchdog_task(void *arg) {
    while (1) {
        if (marlin_connected &&
            (xTaskGetTickCount() * portTICK_PERIOD_MS - last_marlin_reply_ms > 8000)) {
            marlin_connected = false;
            uart_flush(MARLIN_UART_NUM);
            xQueueSend(marlin_cmd_queue, "M105", 0);
        }
        vTaskDelay(pdMS_TO_TICKS(1000));
    }
}

        }
        lvgl_port_unlock();
    }

    // Tasks
    xTaskCreate(marlin_rx_task,      "rx",   4096, NULL, 5, NULL);
    xTaskCreate(marlin_tx_task,      "tx",   3072, NULL, 3, NULL);
    xTaskCreate(status_poller_task,  "poll", 2048, NULL, 2, NULL);
    xTaskCreate(watchdog_task,       "wd",   2048, NULL, 1, NULL);

    lv_timer_create(update_clock_labels_timer, 1000, NULL);

    // Initial link
    send_gcode("M155 S2");
    vTaskDelay(pdMS_TO_TICKS(200));
    send_gcode("M20");  // ← ADDED: request file list on boot
}

Reviewed LVGL 9 list layout and event handling
The "grouped" look and lack of selection are both caused by the same issue: lv_list_add_text creates plain labels, not buttons. Labels aren't interactive, so they don't fire click events and don't get the button styling that makes items look like a proper list.

Also, lv_list is deprecated in LVGL 9.6 — it's just a flex column under the hood. The fix is to create actual buttons inside the container. 

EEZ Studio layout check
Make sure your file_browser_list object in EEZ Studio has:

Layout: Flex
Flex Flow: Column (not Row, not Grid)
Wrap: Off
Scrollable: Yes (vertical)
If it's set to Grid or Row, that's why items appear "grouped" in multiple columns.

Verifying LV_BUBBLE_CLICKED availability in LVGL 8.4
The click events on the buttons aren't reaching the parent container. In LVGL 8, you must explicitly enable event bubbling on each child button with LV_OBJ_FLAG_EVENT_BUBBLE. 

Fix in esp32_s3.c — add one line after creating each button
if (strncmp(item.line, "LIST_ADD:", 9) == 0) {
    if (objects.file_browser_list) {
        lv_obj_t *btn = lv_btn_create(objects.file_browser_list);

        lv_obj_center(label);
    }
}

Fix in main.cpp — use lv_event_get_target not lv_event_get_current_target
When an event bubbles up, lv_event_get_target(e) gives you the original button that was clicked, while lv_event_get_current_target(e) gives you the container (where the callback is attached). You need the button to read its label:

static void file_browser_click_cb(lv_event_t *e) {
    lv_obj_t *btn = lv_event_get_target(e);  // the actual button clicked
    if (!btn) return;
    lv_obj_t *label = lv_obj_get_child(btn, 0);
    if (!label) return;
    const char *filename = lv_label_get_text(label);
    if (!filename || strlen(filename) == 0) return;

    show_print_confirmation(filename);
}

That's it — the LV_OBJ_FLAG_EVENT_BUBBLE flag is what makes the parent's LV_EVENT_CLICKED callback fire when a child button is tapped.

Identified dangling pointer in btn_labels array
The crash is a use-after-scope bug. In LVGL 8, lv_msgbox_create stores a pointer to your btn_labels array — it does NOT copy it. Your array is a local variable that dies when show_print_confirmation returns, so when LVGL later lays out the btnmatrix, it reads freed stack memory → null deref → crash.

Fix in main.cpp
Make the button labels static so they live for the entire program:

static void show_print_confirmation(const char *filename) {
    snprintf(pending_filename, sizeof(pending_filename), "%s", filename);

    static const char *btn_labels[] = {"Print", "Cancel", NULL};  // ← static
    lv_obj_t *mbox = lv_msgbox_create(NULL, "Confirm Print", filename, btn_labels, false);
    lv_obj_add_event_cb(mbox, confirm_print_cb, LV_EVENT_VALUE_CHANGED, NULL);
}

The only change: const char *btn_labels[] → static const char *btn_labels[].

That's it. The static keyword puts the array in the data segment instead of the stack, so the pointer remains valid for the lifetime of the msgbox.

Analyzed LVGL 8 msgbox touch propagation issue
This is a classic LVGL 8 issue: when the msgbox is deleted during a touch event, the touch release propagates through to whatever button is underneath (your "Set Time" button). The fix is to add a full-screen blocking overlay behind the msgbox so no clicks reach the main screen while the dialog is open.

Fix in main.cpp
Replace show_print_confirmation and confirm_print_cb with this:

static lv_obj_t *modal_overlay = NULL;

static void confirm_print_cb(lv_event_t *e) {
    lv_obj_t *mbox = lv_event_get_current_target(e);
    uint16_t btn_idx = lv_msgbox_get_active_btn(mbox);

    if (btn_idx == 0) {  // "Print"
        char cmd[80];
        snprintf(cmd, sizeof(cmd), "M23 %s", pending_filename);
        send_gcode(cmd);
        isPrinting = true;
    }

    if (mbox) lv_obj_del(mbox);
    if (modal_overlay) {
        lv_obj_del(modal_overlay);
        modal_overlay = NULL;
    }
}

static void show_print_confirmation(const char *filename) {
    snprintf(pending_filename, sizeof(pending_filename), "%s", filename);

    // Full-screen overlay to block clicks to the main screen
    lv_obj_t *scr = lv_obj_get_screen(NULL);
    modal_overlay = lv_obj_create(scr);
    lv_obj_set_size(modal_overlay, LV_PCT(100), LV_PCT(100));
    lv_obj_set_pos(modal_overlay, 0, 0);
    lv_obj_set_style_bg_opa(modal_overlay, LV_OPA_50, 0);
    lv_obj_set_style_bg_color(modal_overlay, lv_color_black(), 0);
    lv_obj_add_flag(modal_overlay, LV_OBJ_FLAG_CLICKABLE);  // absorbs all touches

    static const char *btn_labels[] = {"Print", "Cancel", NULL};
    lv_obj_t *mbox = lv_msgbox_create(modal_overlay, "Confirm Print", filename, btn_labels, false);
    lv_obj_center(mbox);
    lv_obj_add_event_cb(mbox, confirm_print_cb, LV_EVENT_VALUE_CHANGED, NULL);
}

What changed:

Before	After	Why
lv_msgbox_create(NULL, ...)	lv_msgbox_create(modal_overlay, ...)	Msgbox is now a child of the overlay, not the screen
No overlay	Full-screen clickable overlay behind msgbox	Absorbs all touch events so nothing on the main screen gets clicked
Delete only mbox	Delete both mbox and overlay	Cleans up both when dialog closes

The overlay's LV_OBJ_FLAG_CLICKABLE flag is the key — it makes the overlay "eat" all touches, so your "Set Time" button (or any other button) underneath can never receive a click while the dialog is open.

// ─── Marlin line parser ──────────────────────────────────────────────────────

void parse_marlin_line(const char *line) {
    if (line == NULL || strlen(line) == 0) return;

    // Temperature updates
    float hcur = -1, bcur = -1, dcur = -1;
    const char *tPos = strstr(line, "T:");
    if (tPos) sscanf(tPos, "T:%f", &hcur);
    const char *bPos = strstr(line, "B:");
    if (bPos) sscanf(bPos, "B:%f", &bcur);
    const char *cPos = strstr(line, "C:");
    if (cPos) sscanf(cPos, "C:%f", &dcur);

    char buf[16];
    if (hcur >= 0 && objects.nozzle_temp_main_screen) {
        snprintf(buf, sizeof(buf), "%.0f°", hcur);
        lv_label_set_text(objects.nozzle_temp_main_screen, buf);
    }
    if (bcur >= 0 && objects.bed_temp_main_screen) {
        snprintf(buf, sizeof(buf), "%.0f°", bcur);
        lv_label_set_text(objects.bed_temp_main_screen, buf);
    }
    if (dcur >= 0 && objects.drivers_temp_main_screen) {
        snprintf(buf, sizeof(buf), "%.0f°", dcur);
        lv_label_set_text(objects.drivers_temp_main_screen, buf);
    }

    // M20 file list parsing
    if (strcmp(line, "Begin file list") == 0) {
        file_count = 0;
        capturing_files = true;
        return;
    }
    if (strcmp(line, "End file list") == 0) {
        capturing_files = false;
        update_file_browser_ui();
        return;
    }
    if (capturing_files && file_count < MAX_FILES) {
        const char *name = line;

        char *colon = strchr(line, ':');
        if (colon && (colon - line) < 3) {
            name = colon + 1;
        }

        char *tmp = (char *)name;
        char *last_space = strrchr(tmp, ' ');
        if (last_space) {
            char *size_str = last_space + 1;
            bool is_size = (size_str[0] != '\0');
            for (char *p = size_str; *p; p++) {
                if (*p < '0' || *p > '9') { is_size = false; break; }
            }
            if (is_size) *last_space = '\0';
        }

        snprintf(file_list[file_count], FILE_NAME_LEN, "%.63s", tmp);
        file_count++;
    }

    // ← ADDED: detect print start / completion from Marlin
    if (strstr(line, "Start SD print job:")) {
        isPrinting = true;
    }
    if (strstr(line, "Done printing file")) {
        isPrinting = false;
    }
}

// ─── Tasks ───────────────────────────────────────────────────────────────────

static void marlin_tx_task(void *arg) {
    char cmd[MAX_GCODE_LEN];
    while (1) {
        if (xQueueReceive(marlin_cmd_queue, cmd, portMAX_DELAY)) {
            size_t len = strlen(cmd);
            if (cmd[len - 1] != '\n') {
                cmd[len] = '\n';
                len++;
            }
            uart_write_bytes(MARLIN_UART_NUM, cmd, len);
            uart_wait_tx_done(MARLIN_UART_NUM, pdMS_TO_TICKS(100));
        }
    }
}

static void marlin_rx_task(void *arg) {
    uint8_t rx_buf[256];
    static char line_accumulator[512];
    static size_t line_pos = 0;

    while (1) {
        int len = uart_read_bytes(MARLIN_UART_NUM, rx_buf, sizeof(rx_buf) - 1, pdMS_TO_TICKS(20));
        if (len > 0) {
            last_marlin_reply_ms = xTaskGetTickCount() * portTICK_PERIOD_MS;
            marlin_connected = true;

            for (int i = 0; i < len; i++) {
                char c = (char)rx_buf[i];
                if (c == '\n') {
                    line_accumulator[line_pos] = '\0';
                    if (line_pos > 0) {
                        parse_marlin_line(line_accumulator);

                        if (strstr(line_accumulator, "T:") == NULL &&
                            strcmp(line_accumulator, "ok") != 0 &&
                            strncmp(line_accumulator, "ok ", 3) != 0 &&
                            !capturing_files &&
                            strcmp(line_accumulator, "Begin file list") != 0 &&
                            strcmp(line_accumulator, "End file list") != 0) {
                            append_print_log(line_accumulator);
                        }
                    }
                    line_pos = 0;
                } else if (c != '\r' && line_pos < sizeof(line_accumulator) - 1) {
                    line_accumulator[line_pos++] = c;
                }
            }
        }
        vTaskDelay(pdMS_TO_TICKS(10));
    }
}

static void status_poller_task(void *arg) {
    while (1) {
        if (isPrinting) send_gcode("M27");
        vTaskDelay(pdMS_TO_TICKS(2000));
    }
}

static void watchdog_task(void *arg) {
    while (1) {
        if (marlin_connected &&
            (xTaskGetTickCount() * portTICK_PERIOD_MS - last_marlin_reply_ms > 8000)) {
            marlin_connected = false;
            uart_flush(MARLIN_UART_NUM);
            xQueueSend(marlin_cmd_queue, "M105", 0);
        }
        vTaskDelay(pdMS_TO_TICKS(1000));
    }
}

Diagnosed SD print stuck at byte 0
Yes, it is reading and executing the G-code. The file is open (7.8 MB confirmed). Your printer is almost certainly stuck at:

M190 S70    ; Wait for bed to reach 70°C
M109 S245   ; Wait for nozzle to reach 118°C

These are blocking commands — Marlin halts file execution until the temperatures are reached. The nozzle needs to heat from ambient to 245°F (118°C), which takes 5–15 minutes depending on your hotend.

The "byte 0" in the M27 report is misleading — some Marlin versions don't advance the byte counter while blocked on a temperature wait.

How to confirm it's working
Watch the temperature labels on your HMI (the T: / B: values from M155 S2). If the nozzle temp is climbing toward 118°C and the bed toward 70°C, the print is progressing normally. Once both reach target, the byte counter will jump forward and you'll see the bed move (G28, purge lines, brim).

You don't need to send G28 before M23
Your G-code file already handles everything in the correct order:

Set temps (non-blocking) → M140, M104
Home → G28
Wait for temps → M190, M109
Purge → prime lines
Print
