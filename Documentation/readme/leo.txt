Buzzer Configuration for CrowPanel DIS02170A V1.4 800×480 (Advansed-Series) ESP32-S3-WROOM-1-N16R8 idf 5.5.1

Quick Answer
For the CrowPanel Advance 7" HMI (DIS02170A) version V1.4 running ESP-IDF 5.5.1, the buzzer is controlled via the STC8 microcontroller (I2C address 0x30) rather than direct GPIO toggling used in V1.0.

To activate the buzzer, send the command 0xF6 (decimal 246) to the STC8. To deactivate it, send 0xF7 (decimal 247). 

Implementation Details
Communication: Use the I2C interface (SCL: GPIO_NUM_16, SDA: GPIO_NUM_15) to communicate with the STC8 helper chip.
Command Structure: The STC8 acts as a peripheral manager. The specific byte values 246 (0xF6) and 247 (0xF7) are the standard commands for activating and deactivating the buzzer on V1.2+ revisions, including V1.4. 
Code Example (ESP-IDF):
#include <driver/i2c.h>

#define STC8_I2C_ADDR 0x30
#define I2C_SCL GPIO_NUM_16
#define I2C_SDA GPIO_NUM_15

void buzzer_on() {
    i2c_cmd_handle_t cmd = i2c_cmd_link_create();
    i2c_master_start(cmd);
    i2c_master_write_byte(cmd, (STC8_I2C_ADDR << 1) | I2C_MASTER_WRITE, true);
    i2c_master_write_byte(cmd, 0xF6, true); // Activate buzzer
    i2c_master_stop(cmd);
    i2c_master_cmd_begin(I2C_NUM_0, cmd, pdMS_TO_TICKS(1000));
    i2c_cmd_link_delete(cmd);
}

void buzzer_off() {
    i2c_cmd_handle_t cmd = i2c_cmd_link_create();
    i2c_master_start(cmd);
    i2c_master_write_byte(cmd, (STC8_I2C_ADDR << 1) | I2C_MASTER_WRITE, true);
    i2c_master_write_byte(cmd, 0xF7, true); // Deactivate buzzer
    i2c_master_stop(cmd);
    i2c_master_cmd_begin(I2C_NUM_0, cmd, pdMS_TO_TICKS(1000));
    i2c_cmd_link_delete(cmd);
}

Note that V1.4 shares the same hardware control logic as V1.3, with only the button component updated in the V1.4 release. Ensure your I2C driver is initialized at 400 kHz speed before issuing these commands.

Understood — you're using the new driver/i2c_master.h API (not the legacy driver/i2c.h).  Here's the corrected buzzer code for that:

#include "driver/i2c_master.h"
#include "driver/ledc.h"
#include "driver/gpio.h"

#define STC8_I2C_ADDR  0x30
#define I2C_SCL_IO     GPIO_NUM_16
#define I2C_SDA_IO     GPIO_NUM_15
#define I2C_PORT       I2C_NUM_0

static i2c_master_bus_handle_t i2c_bus = NULL;
static i2c_master_dev_handle_t stc8_dev = NULL;

esp_err_t i2c_stc8_init(void)
{
    i2c_master_bus_config_t bus_cfg = {
        .i2c_port = I2C_PORT,
        .scl_io_num = I2C_SCL_IO,
        .sda_io_num = I2C_SDA_IO,
        .clk_source = I2C_CLK_SRC_DEFAULT,
        .glitch_ignore_cnt = 7,
        .flags.enable_internal_pullup = true,
    };
    ESP_ERROR_CHECK(i2c_new_master_bus(&bus_cfg, &i2c_bus));

    i2c_device_config_t dev_cfg = {
        .dev_addr_length = I2C_ADDR_BIT_LEN_7,
        .device_address = STC8_I2C_ADDR,
        .scl_speed_hz = 400000,
    };
    ESP_ERROR_CHECK(i2c_master_bus_add_device(i2c_bus, &dev_cfg, &stc8_dev));
    return ESP_OK;
}

void buzzer_on(void)
{
    uint8_t cmd = 0xF6;
    ESP_ERROR_CHECK(i2c_master_transmit(stc8_dev, &cmd, 1, 1000));
}

void buzzer_off(void)
{
    uint8_t cmd = 0xF7;
    ESP_ERROR_CHECK(i2c_master_transmit(stc8_dev, &cmd, 1, 1000));
}

Key differences from the legacy API
Legacy (driver/i2c.h)	New (driver/i2c_master.h)
i2c_param_config() + i2c_driver_install()	i2c_new_master_bus()
i2c_cmd_link_create() / i2c_master_start() / i2c_master_write_byte() / i2c_master_stop() / i2c_master_cmd_begin()	Single call: i2c_master_transmit(dev, buf, len, timeout_ms)
Address shifted manually (addr << 1)	Address passed as 7-bit value in device_address

The new driver handles the start/stop/address framing internally — you just supply the data payload (the 0xF6/0xF7 command byte) and the device handle takes care of the rest.

Make sure your CMakeLists.txt has esp_driver_i2c in REQUIRES (not the old driver component).

Verified ESP-IDF project structure
No — you don't place anything in a "drivers folder." The driver/ directory is part of the ESP-IDF SDK itself and is read-only. You never modify it.

Here's what to do:

The #include lines (driver/i2c_master.h, driver/ledc.h, driver/gpio.h) — these are SDK headers already available in your toolchain. Just include them in your source file. Nothing to copy.
The actual code (the i2c_stc8_init(), buzzer_on(), buzzer_off() functions) — put it in your main/ directory, e.g. main/buzzer.c with a matching main/buzzer.h.
main/CMakeLists.txt — make sure esp_driver_i2c is listed:
idf_component_register(SRCS "main.c" "buzzer.c"
                       INCLUDE_DIRS "."
                       REQUIRES esp_driver_i2c esp_driver_gpio esp_driver_ledc)

That's it — the SDK headers resolve automatically, and your code lives in main/.

Identified I2C bus conflict and fix
The I2C bus is already initialized by your stc8h1k28 component. You can't call i2c_new_master_bus() a second time on the same port.

Fix: get the existing bus handle instead of creating a new one:

#include "driver/i2c_master.h"

#define STC8_I2C_ADDR  0x30
#define I2C_PORT       I2C_NUM_0

static i2c_master_bus_handle_t i2c_bus = NULL;
static i2c_master_dev_handle_t stc8_dev = NULL;

esp_err_t i2c_stc8_init(void)
{
    // Bus is already created by the stc8h1k28 component — just grab it
    ESP_ERROR_CHECK(i2c_get_bus_handle(I2C_PORT, &i2c_bus));

    i2c_device_config_t dev_cfg = {
        .dev_addr_length = I2C_ADDR_BIT_LEN_7,
        .device_address   = STC8_I2C_ADDR,
        .scl_speed_hz     = 400000,
    };
    ESP_ERROR_CHECK(i2c_master_bus_add_device(i2c_bus, &dev_cfg, &stc8_dev));
    return ESP_OK;
}

void buzzer_on(void)
{
    uint8_t cmd = 0xF6;
    ESP_ERROR_CHECK(i2c_master_transmit(stc8_dev, &cmd, 1, 1000));
}

void buzzer_off(void)
{
    uint8_t cmd = 0xF7;
    ESP_ERROR_CHECK(i2c_master_transmit(stc8_dev, &cmd, 1, 1000));
}

The only change: i2c_new_master_bus(...) → i2c_get_bus_handle(I2C_PORT, &i2c_bus).

Even better — check if your stc8h1k28 component already exposes the device handle (e.g. via a header or a getter function). If it does, skip i2c_stc8_init() entirely and just use that handle directly in buzzer_on()/buzzer_off().

esp_err_t stc8h1k28_init(i2c_master_bus_handle_t bus_handle, stc8h1k28_handle_t *extender_handle, const i2c_device_config_t *config) {
    ESP_RETURN_ON_FALSE(bus_handle && extender_handle && config, ESP_ERR_INVALID_ARG, TAG, "Invalid arguments");

    stc8h1k28_dev_t *dev = calloc(1, sizeof(stc8h1k28_dev_t));

    // Initialize I2C device

        TAG, "Failed to add I2C device");

    *extender_handle = dev;
    ESP_LOGI(TAG, "stc8h1k28 initialized successfully");
    return ESP_OK;
}

    if (dev) {

        free(dev);
    }
}

Explained LVGL task priority impact
Not related to the display artifacts. It controls how aggressively the LVGL task preempts other tasks when rendering UI updates.

0 = lowest priority — other tasks (serial, USB, etc.) can delay UI refresh
2 = moderate — UI stays responsive without starving critical drivers
Keep it at 2. It won't cause or fix the stretching/red lines. If the UI ever feels sluggish under load (e.g., while Marlin is streaming G-code over USB), you can bump it to 3 or 4, but 2 is a safe default for this board.

/**
 * @brief Initialize LVGL Library
 *
 * This function initializes the LVGL library, allocates separate draw buffers from PSRAM,
 * registers the display driver and input device driver to LVGL, creates a semaphore for
 * LVGL synchronization, and starts the LVGL port task.
 *
 * @param[in] panel_handle Handle to the LCD panel associated with LVGL.
 * @param[in] touch_handle Handle to the touchpad device associated with LVGL.
 */
void init_lvgl(esp_lcd_panel_handle_t panel_handle, esp_lcd_touch_handle_t touch_handle) {

    ESP_LOGI(TAG, "Initialize LVGL library");
	
	
    // Initialize LVGL core first
    lv_init();

    // Create mutex after lv_init
    lvgl_mux = xSemaphoreCreateRecursiveMutex(); // Recommended for LVGL re-entrant locking
    if (lvgl_mux == NULL) {
        ESP_LOGE(TAG, "Failed to create LVGL mutex!");
        return;
    }

    ESP_ERROR_CHECK(tick_init());

    static lv_disp_draw_buf_t disp_buf;
    static lv_disp_drv_t disp_drv;
    
    void *buf1 = NULL;
    void *buf2 = NULL;

    ESP_LOGI(TAG, "Use frame buffers as LVGL draw buffers");
    ESP_ERROR_CHECK(esp_lcd_rgb_panel_get_frame_buffer(panel_handle, 2, &buf1, &buf2));
    // initialize LVGL draw buffers
    lv_disp_draw_buf_init(&disp_buf, buf1, buf2, LCD_H_RES * LCD_V_RES);

    ESP_LOGI(TAG, "Register display driver to LVGL");
    lv_disp_drv_init(&disp_drv);
    disp_drv.hor_res = LCD_H_RES;
    disp_drv.ver_res = LCD_V_RES;
    disp_drv.flush_cb = lvgl_flush_cb;
    disp_drv.draw_buf = &disp_buf;
    disp_drv.user_data = panel_handle;
    disp_drv.full_refresh = true; // the full_refresh mode can maintain the synchronization between the two frame buffers
    // Register and set as default display context
	lv_disp_t *disp = lv_disp_drv_register(&disp_drv);
	lv_disp_set_default(disp);

    ESP_LOGI(TAG, "Register input device driver to LVGL");
    static lv_indev_drv_t indev_drv;

// ---- Affine solve + verify  ----

typedef struct {
    float a, b, c;  // x_screen = a*rx + b*ry + c
    float d, e, f;  // y_screen = d*rx + e*ry + f
} affine_t;

// Solve 3x3 system M*x = r using Cramer's rule
static bool solve3(const float m[9], const float r[3], float out[3]) {
    float m00=m[0], m01=m[1], m02=m[2];
    float m10=m[3], m11=m[4], m12=m[5];
    float m20=m[6], m21=m[7], m22=m[8];
    float r0=r[0], r1=r[1], r2=r[2];

    float det = m00*(m11*m22 - m12*m21)
              - m01*(m10*m22 - m12*m20)
              + m02*(m10*m21 - m11*m20);
    if (fabsf(det) < 1e-6f) return false;

    out[0] = ( r0*(m11*m22 - m12*m21)
             - m01*(r1*m22 - m12*r2)
             + m02*(r1*m21 - m11*r2)) / det;

    out[1] = ( m00*(r1*m22 - m12*r2)
             - r0 *(m10*m22 - m12*m20)
             + m02*(m10*r2 - r1*m20)) / det;

    out[2] = ( m00*(m11*r2 - r1*m21)
             - m01*(m10*r2 - r1*m20)
             + r0 *(m10*m21 - m11*m20)) / det;

    return true;
}

// Compute affine from 3 raw→screen point pairs
static bool affine_from_3(const cal_point_t *pts, affine_t *m) {
    // x system: a*rx + b*ry + c = sx
    float mx[9] = {
        (float)pts[0].tx, (float)pts[0].ty, 1.0f,
        (float)pts[1].tx, (float)pts[1].ty, 1.0f,
        (float)pts[2].tx, (float)pts[2].ty, 1.0f
    };
    float rx[3] = { (float)pts[0].px, (float)pts[1].px, (float)pts[2].px };
    float ry[3] = { (float)pts[0].py, (float)pts[1].py, (float)pts[2].py };
    float sol[3];

    if (!solve3(mx, rx, sol)) return false;
    m->a = sol[0]; m->b = sol[1]; m->c = sol[2];

    // y system: d*rx + e*ry + f = sy
    if (!solve3(mx, ry, sol)) return false;
    m->d = sol[0]; m->e = sol[1]; m->f = sol[2];
    return true;
}

// Apply affine: raw → screen
static void affine_apply(const affine_t *m, float rx, float ry, float *sx, float *sy) {
    *sx = m->a * rx + m->b * ry + m->c;
    *sy = m->d * rx + m->e * ry + m->f;
}

    float d, e, f;  // y_screen = d*rx + e*ry + f
} affine_t;

// Solve 3x3 system M*x = r using Cramer's rule
static bool solve3(const float m[9], const float r[3], float out[3]) {
    float m00=m[0], m01=m[1], m02=m[2];
    float m10=m[3], m11=m[4], m12=m[5];
    float m20=m[6], m21=m[7], m22=m[8];
    float r0=r[0], r1=r[1], r2=r[2];

    float det = m00*(m11*m22 - m12*m21)
              - m01*(m10*m22 - m12*m20)
              + m02*(m10*m21 - m11*m20);
    if (fabsf(det) < 1e-6f) return false;

    out[0] = ( r0*(m11*m22 - m12*m21)
             - m01*(r1*m22 - m12*r2)
             + m02*(r1*m21 - m11*r2)) / det;

    out[1] = ( m00*(r1*m22 - m12*r2)
             - r0 *(m10*m22 - m12*m20)
             + m02*(m10*r2 - r1*m20)) / det;

    out[2] = ( m00*(m11*r2 - r1*m21)
             - m01*(m10*r2 - r1*m20)
             + r0 *(m10*m21 - m11*m20)) / det;

    return true;
}

// Compute affine from 3 raw→screen point pairs
static bool affine_from_3(const cal_point_t *pts, affine_t *m) {
    // x system: a*rx + b*ry + c = sx
    float mx[9] = {
        (float)pts[0].tx, (float)pts[0].ty, 1.0f,
        (float)pts[1].tx, (float)pts[1].ty, 1.0f,
        (float)pts[2].tx, (float)pts[2].ty, 1.0f
    };
    float rx[3] = { (float)pts[0].px, (float)pts[1].px, (float)pts[2].px };
    float ry[3] = { (float)pts[0].py, (float)pts[1].py, (float)pts[2].py };
    float sol[3];

    if (!solve3(mx, rx, sol)) return false;
    m->a = sol[0]; m->b = sol[1]; m->c = sol[2];

    // y system: d*rx + e*ry + f = sy
    if (!solve3(mx, ry, sol)) return false;
    m->d = sol[0]; m->e = sol[1]; m->f = sol[2];
    return true;
}

    *sx = m->a * rx + m->b * ry + m->c;
    *sy = m->d * rx + m->e * ry + m->f;
}

Yes, it's fixed volume. The STC8 just drives the piezo buzzer on or off — there's no PWM or volume register. The I2C interface only exposes two commands: 0xF6 (on) and 0xF7 (off). No in-between.

Your pins are wrong. GPIO 43/44 are the console UART (UART0) — your boot log confirms it: "GPIO 44 and 43 are used as console UART I/O pins." Using them for Marlin serial will kill your debug output. 

The CrowPanel Advance has a dedicated UART1-OUT port on the back:

Port	RX	TX
UART0-OUT	IO44	IO43	← console, don't touch
UART1-OUT	IO19	IO20	← use this for Marlin

Change your defines:

#define MARLIN_RX_PIN     GPIO_NUM_19   // printer TX → ESP32 RX

#define COMMAND_QUEUE_LEN 20

Wiring (BTT TFT header on Octopus 1.1 → CrowPanel UART1-OUT):

BTT TFT header	→	CrowPanel UART1-OUT
TX (printer sends)	→	RX (IO19)
RX (printer receives)	→	TX (IO20)
GND	→	GND
5V	→	5V (if powering from printer)

You don't need the RESET line for basic operation.

void marlin_rx_task(void *arg)
{
    uint8_t buf[128];
    char line[128] = {0};
    int line_pos = 0;

    while (1) {
        int len = uart_read_bytes(MARLIN_UART_NUM, buf, sizeof(buf), pdMS_TO_TICKS(50));
        for (int i = 0; i < len; i++) {
            if (buf[i] == '\n' || buf[i] == '\r') {
                if (line_pos > 0) {
                    line[line_pos] = '\0';
                    parse_marlin_line(line);   // ← your existing parser
                    line_pos = 0;
                }

Memory — a textarea that grows forever will eventually exhaust PSRAM. Add a size cap:
void append_to_log(const char *line) {
    if (!objects.print_status_log_print_screen) return;
    
    // Cap at ~2000 chars to prevent memory bloat
    const char *existing = lv_textarea_get_text(objects.print_status_log_print_screen);
    if (strlen(existing) > 2000) {
        lv_textarea_set_text(objects.print_status_log_print_screen, "");
    }
    
    char buf[256];
    snprintf(buf, sizeof(buf), "%s\n", line);
    lv_textarea_add_text(objects.print_status_log_print_screen, buf);
    lv_textarea_set_cursor_pos(objects.print_status_log_print_screen, LV_TEXTAREA_CURSOR_LAST);

void append_to_log(const char *line) {
    if (!objects.print_status_log_print_screen) return;

        lv_textarea_add_text(objects.print_status_log_print_screen, buf);
        lv_textarea_set_cursor_pos(objects.print_status_log_print_screen, LV_TEXTAREA_CURSOR_LAST);
        lvgl_port_unlock();
    }
}
