void app_main(void)
{
    init_display();
    ESP_LOGI(TAG, "Start LVGL");

    // RTC: read + sync system clock
    struct tm rtc_time = {};
    if (bm8563_get_time(rtc_handle, &rtc_time) == ESP_OK) {
        ESP_LOGI(TAG, "RTC: %04d-%02d-%02d %02d:%02d:%02d",
                 rtc_time.tm_year + 1900, rtc_time.tm_mon + 1, rtc_time.tm_mday,
                 rtc_time.tm_hour, rtc_time.tm_min, rtc_time.tm_sec);

        time_t t = mktime(&rtc_time);
        struct timeval tv = { .tv_sec = t, .tv_usec = 0 };
        settimeofday(&tv, NULL);
        ESP_LOGI(TAG, "System clock synced to RTC");
    } else {
        ESP_LOGE(TAG, "RTC read FAILED");
    }

    // Boot beep
    stc8 = stc8h1k28_get_handle();
    stc8h1k28_buzzer_on(stc8);
    vTaskDelay(pdMS_TO_TICKS(1000));
    stc8h1k28_buzzer_off(stc8);

    if (lvgl_port_lock(portMAX_DELAY)) {
        ui_init();
        lvgl_port_unlock();
        ESP_LOGI(TAG, "UI initialized successfully!");
    } else {
        ESP_LOGE(TAG, "Failed to lock LVGL port for UI initialization");
    }

    lv_timer_create(update_clock_labels_timer, 1000, NULL);
}
