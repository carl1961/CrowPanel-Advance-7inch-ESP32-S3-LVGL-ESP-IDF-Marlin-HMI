#define MARLIN_UART_NUM   UART_NUM_1
#define MARLIN_TX_PIN     GPIO_NUM_20   // ESP32 TX → printer RX
#define MARLIN_RX_PIN     GPIO_NUM_19   // printer TX → ESP32 RX
#define SERIAL_BUF_SIZE   1024
#define MAX_GCODE_LEN     64
#define COMMAND_QUEUE_LEN 20

static stc8h1k28_handle_t stc8 = NULL; 
static const char* TAG = "MAIN";

extern SemaphoreHandle_t lvgl_mux;

void update_clock_labels_timer(lv_timer_t *timer);

bool queue_gcode_command(const char* gcode) {
    if (!marlin_cmd_queue) return false;
    char cmdbuf[MAX_GCODE_LEN];
    snprintf(cmdbuf, sizeof(cmdbuf), "%s\n", gcode);
    return xQueueSend(marlin_cmd_queue, cmdbuf, 0) == pdPASS;
}

static void marlin_tx_task(void *arg) {
    char cmd[MAX_GCODE_LEN];
    while (1) {
        if (xQueueReceive(marlin_cmd_queue, cmd, portMAX_DELAY)) {
            uart_write_bytes(MARLIN_UART_NUM, cmd, strlen(cmd));
            uart_wait_tx_done(MARLIN_UART_NUM, pdMS_TO_TICKS(100));
        }
    }
}

static void marlin_rx_task(void *arg) {
    uint8_t rx_buf[128];
    static char line_accumulator[256];
    static size_t line_pos = 0;

    while (1) {
        int len = uart_read_bytes(MARLIN_UART_NUM, rx_buf, sizeof(rx_buf) - 1, pdMS_TO_TICKS(20));
        if (len > 0) {
            last_marlin_reply_ms = xTaskGetTickCount() * portTICK_PERIOD_MS;
            marlin_connected = true;

            for (int i = 0; i < len; i++) {
                char c = (char)rx_buf[i];
                if (c == '\n') {
                    line_accumulator[line_pos] = '\0';
                    if (line_pos > 0) {
                        strncpy(last_marlin_line, line_accumulator, sizeof(last_marlin_line) - 1);
                        new_line_ready = true;
                    }
                    line_pos = 0;
                } else if (c != '\r' && line_pos < sizeof(line_accumulator) - 1) {
                    line_accumulator[line_pos++] = c;
                }
            }
        }
        vTaskDelay(pdMS_TO_TICKS(10));
    }
}

static bool force_marlin_link(void) {
    const int bauds[] = {115200, 115200};
    const char *cmds[] = {"M115\n", "M105\n"};
    for (int b = 0; b < 2; b++) {
        uart_set_baudrate(MARLIN_UART_NUM, bauds[b]);
        uart_flush(MARLIN_UART_NUM);
        vTaskDelay(pdMS_TO_TICKS(100));
        for (int c = 0; c < 2; c++) {
            uart_write_bytes(MARLIN_UART_NUM, cmds[c], strlen(cmds[c]));
            vTaskDelay(pdMS_TO_TICKS(400));
            uint8_t rbuf[256];
            int len = uart_read_bytes(MARLIN_UART_NUM, rbuf, sizeof(rbuf) - 1, pdMS_TO_TICKS(300));
            if (len > 0) {
                marlin_connected = true;
                return true;
            }
        }
    }
    return false;
}

static void status_poller_task(void *arg) {
    while (1) {
        queue_gcode_command("M105");
        if (isPrinting) queue_gcode_command("M27");
        vTaskDelay(pdMS_TO_TICKS(2000));
    }
}

static void watchdog_task(void *arg) {
    while (1) {
        if (marlin_connected && (xTaskGetTickCount() * portTICK_PERIOD_MS - last_marlin_reply_ms > 8000)) {
            marlin_connected = false;
            force_marlin_link();
        }
        vTaskDelay(pdMS_TO_TICKS(1000));
    }
}

extern "C" void send_gcode(const char* command) {
    if (!command) return;
    ESP_LOGI("GCODE_UI", ">>> %s", command);
    queue_gcode_command(command);
    if (ui_LabelPanelStatus) {
        lv_label_set_text(ui_LabelPanelStatus, command);
    }
}




void app_main(void)
{
    init_display();
    ESP_LOGI(TAG, "Start LVGL");

    // RTC: read + sync system clock
    struct tm rtc_time = {};
    if (bm8563_get_time(rtc_handle, &rtc_time) == ESP_OK) {
        ESP_LOGI(TAG, "RTC: %04d-%02d-%02d %02d:%02d:%02d",
                 rtc_time.tm_year + 1900, rtc_time.tm_mon + 1, rtc_time.tm_mday,
                 rtc_time.tm_hour, rtc_time.tm_min, rtc_time.tm_sec);
        time_t t = mktime(&rtc_time);
        struct timeval tv = { .tv_sec = t, .tv_usec = 0 };
        settimeofday(&tv, NULL);
    }

    // Boot beep
    stc8 = stc8h1k28_get_handle();
    stc8h1k28_buzzer_on(stc8);
    vTaskDelay(pdMS_TO_TICKS(200));
    stc8h1k28_buzzer_off(stc8);

    // UART init (unchanged from your old code)
    uart_config_t uart_config = {
        .baud_rate = 115200,
        .data_bits = UART_DATA_8_BITS,
        .parity = UART_PARITY_DISABLE,
        .stop_bits = UART_STOP_BITS_1,
        .flow_ctrl = UART_HW_FLOWCTRL_DISABLE,
        .source_clk = UART_SCLK_DEFAULT,   // ← was UART_SCLK_APB, either works
    };
    uart_param_config(MARLIN_UART_NUM, &uart_config);
    uart_set_pin(MARLIN_UART_NUM, MARLIN_TX_PIN, MARLIN_RX_PIN, UART_PIN_NO_CHANGE, UART_PIN_NO_CHANGE);
    uart_driver_install(MARLIN_UART_NUM, SERIAL_BUF_SIZE * 2, 0, 0, NULL, 0);

    // Create tasks
    marlin_cmd_queue = xQueueCreate(COMMAND_QUEUE_LEN, MAX_GCODE_LEN);
    xTaskCreate(marlin_tx_task, "tx", 3072, NULL, 3, NULL);
    xTaskCreate(marlin_rx_task, "rx", 4096, NULL, 5, NULL);
    xTaskCreate(status_poller_task, "poll", 2048, NULL, 2, NULL);
    xTaskCreate(watchdog_task, "wd", 2048, NULL, 1, NULL);

    // UI
    if (lvgl_port_lock(portMAX_DELAY)) {
        ui_init();
        lvgl_port_unlock();
    }

    lv_timer_create(update_clock_labels_timer, 1000, NULL);
    force_marlin_link();

    // No while(1) — LVGL task + UART tasks run independently
}   

void parse_marlin_line(const char* line) {
    if (line == nullptr || strlen(line) == 0) return;

    float hcur = -1, bcur = -1, dcur = -1;

    // T0 (hotend) — reported as "T:" or "T0:"
    const char *t0 = strstr(line, "T:");
    if (!t0) t0 = strstr(line, "T0:");
    if (t0) sscanf(t0, "T:%f", &hcur);

    // T1 (bed)
    const char *t1 = strstr(line, "T1:");
    if (t1) sscanf(t1, "T1:%f", &bcur);

    // T2 (drivers)
    const char *t2 = strstr(line, "T2:");
    if (t2) sscanf(t2, "T2:%f", &dcur);

    char buf[16];

    if (hcur >= 0 && objects.nozzle_temp_main_screen) {
        snprintf(buf, sizeof(buf), "%.0f°", hcur);
        lv_label_set_text(objects.nozzle_temp_main_screen, buf);
    }

    if (bcur >= 0 && objects.bed_temp_main_screen) {
        snprintf(buf, sizeof(buf), "%.0f°", bcur);
        lv_label_set_text(objects.bed_temp_main_screen, buf);
    }

    if (dcur >= 0 && objects.drivers_temp_main_screen) {
        snprintf(buf, sizeof(buf), "%.0f°", dcur);
        lv_label_set_text(objects.drivers_temp_main_screen, buf);
    }
}   


   // --- SD PRINT PROGRESS ---
    if (strstr(line, "SD printing byte") != nullptr) {
        long long current_byte = 0, total_byte = 0;
        if (sscanf(line, "SD printing byte %lld/%lld", &current_byte, &total_byte) == 2 && total_byte > 0) {
            int percent = (int)((current_byte * 100) / total_byte);
            percent = (percent < 0) ? 0 : ((percent > 100) ? 100 : percent);
            if (ui_ArcOverall) lv_arc_set_value(ui_ArcOverall, percent);
            if (ui_LabelBigPercent) lv_label_set_text_fmt(ui_LabelBigPercent, "%d%%", percent);
        }
    }

    // --- PID TUNING ---
    if (strstr(line, "PID Autotune finished") != nullptr && ui_LabelPIDStatus) {
        lv_label_set_text(ui_LabelPIDStatus, "PID Autotune Finished!");
    }
}
